package "Capa Web / HTTP (Controllers)" {
    class AuthController <<Controller>>
    class ProductoController <<Controller>>
    class CarritoController <<Controller>>
    class PedidoController <<Controller>>
    class UsuarioController <<Controller>>
    class CategoriaController <<Controller>>
    class EstadisticaController <<Controller>>
    
    note right of ProductoController: Reciben las peticiones del frontend\n(GET, POST, PUT, DELETE)
    note right of AuthController: Registro y login.\nDevuelve el token JWT + datos del usuario.
    note right of EstadisticaController: Endpoints del dashboard, solo accesibles\npara usuarios con rol ADMIN
}

package "Capa de Negocio (Services)" {
    class ProductoService <<Service>>
    class CarritoService <<Service>>
    class PedidoService <<Service>>
    class UsuarioService <<Service>>
    class CategoriaService <<Service>>
    class EstadisticaService <<Service>>
    
    note right of CarritoService: Contienen la lógica, validan reglas\ny agrupan a las entidades hijas.
    note right of EstadisticaService: Calcula métricas del dashboard.\nPor regla de Agregados (regla 29) NO inyecta\nrepos ajenos: llama a los Services de cada\nagregado (PedidoService, ProductoService, etc.).
}

package "Capa de Configuracion (Seguridad)" {
    class JwtService <<Config>>
    class SecurityConfig <<Config>>
    class CarritoAbandonadoScheduler <<Config>>
    
    note right of JwtService: Genera y valida tokens JWT\n(HS256, Nimbus JOSE).
    note right of SecurityConfig: Configura rutas publicas, autenticadas\ny /api/estadisticas solo ADMIN.
    note right of CarritoAbandonadoScheduler: Scheduler diario (@Scheduled):\nmarca carritos ACTIVO viejos como ABANDONADO\n(propiedad app.carrito.abandonado-dias).
}

package "Capa de Acceso a Datos (Repositories)" {
    package "Agregado Producto" {

        class ColorRepository <<Repository>>
        class TalleRepository <<Repository>>
        class ProductoRepository <<Repository>>

    }
    
    package "Agregado Carrito" {

    }
    
    package "Agregado Pedido" {
        class PedidoRepository <<Repository>>
        class PedidoDetalleRepository <<Repository>>
    }
    
    package "Agregado Usuario" {
        class UsuarioRepository <<Repository>>
    }
}
